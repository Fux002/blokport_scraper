# Optional image-processing extras for the scraped-photo pipeline
# (stone_pipeline/io/image_processing.py). The FAITHFUL enhancement + downscale chain
# needs only OpenCV/numpy/Pillow (core deps). These extras power the GPU steps:
#   - enhancement engine "esrgan": Real-ESRGAN via spandrel
#   - de-watermark: SDXL-inpaint via diffusers (reconstructs the mark's footprint)
# Install on the GPU image-processing container; the core pipeline does not need them.
#
#   pip install -r stone_pipeline/requirements-imageproc.txt
#
# History: Florence-2 (open-vocab detection) and LaMa (plain inpaint) were both dropped —
# Florence missed the faint logo; LaMa smeared patterned stone. SDXL-inpaint replaces LaMa.
# Runs fp16 on CUDA (fast) / fp32 on MPS/CPU. GPU strongly recommended for a full batch.

pillow>=10              # modern cp312 wheel; stops pip backtracking to an uncompilable sdist
torch>=2.2
torchvision>=0.17
spandrel==0.4.2         # loads Real-ESRGAN (RRDBNet) weights for the enhancement engine (pinned)
# The diffusers stack MUST match the GPU base image's torch (2.4.1). Newer diffusers/
# transformers register custom ops with PEP-604 `X | None` type hints that torch 2.4.1's
# torch.library.infer_schema cannot parse -> `import diffusers.models.autoencoders.autoencoder_kl`
# fails and de-watermark silently disables. These 2024-era pins are the last set that imports
# cleanly on torch 2.4.1. (If the base torch is bumped, bump these together.)
diffusers==0.30.0       # SDXL-inpaint pipeline for de-watermarking
transformers==4.44.2    # SDXL text encoders (CLIP) + the non-stone image classifier (CLIPModel); import-fix pin
accelerate==0.33.0      # diffusers device/dtype placement
safetensors==0.4.4      # weights format
