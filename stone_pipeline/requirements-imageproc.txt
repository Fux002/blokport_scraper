# Optional image-processing extras for the scraped-photo pipeline
# (stone_pipeline/io/image_processing.py). The FAITHFUL enhancement + downscale chain
# needs only OpenCV/numpy/Pillow (core deps). These extras power the GPU + de-watermark steps:
#   - enhancement engine "esrgan": Real-ESRGAN via spandrel (local GPU)
#   - non-stone image classifier: CLIP via transformers (local GPU)
#   - de-watermark: FAL FLUX Fill via fal-client (HOSTED -- no local model, needs FAL_KEY + network)
# Install on the GPU image-processing container; the core pipeline does not need them.
#
#   pip install -r stone_pipeline/requirements-imageproc.txt
#
# History: de-watermark was Florence-2+LaMa, then SDXL-inpaint (diffusers). SDXL regenerated a whole
# square footprint and left a visible "box" on veined stone, and pinned the heavy diffusers stack to
# torch 2.4.1. Replaced by FAL FLUX Fill: it fills only the masked logo (composited back so the rest is
# byte-identical), billed per megapixel on the cropped logo, and drops diffusers + the baked SDXL weights.

pillow>=10              # modern cp312 wheel; stops pip backtracking to an uncompilable sdist
torch>=2.2
torchvision>=0.17
spandrel==0.4.2         # loads Real-ESRGAN (RRDBNet) weights for the enhancement engine (pinned)
transformers==4.44.2    # the non-stone image classifier (CLIPModel); pin imports cleanly on torch 2.4.1
safetensors==0.4.4      # weights format (CLIP)
fal-client>=0.4         # FAL FLUX Fill de-watermark client (hosted inpainting; FAL_KEY from env)
requests>=2.31          # download the FAL result
ben2                    # BEN2 background-removal for VARIANT TEXTURES (image_pipeline/rb_images.py), needed
                        # by the on-demand GPU texture job (RUN_MODE=generate-textures). Pulls its model from
                        # HF at first use (rb_images: BEN_Base.from_pretrained); a pinned build-time bake --
                        # like the ESRGAN/CLIP weights -- is a follow-up hardening, not required to run.
